📄 vite.config.ts
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import tailwindcss from "@tailwindcss/vite";

export default defineConfig({
  plugins: [
    react(),
    tailwindcss(),  // ✅ 一行搞定 v4 集成
  ],
});
📄 src/index.css （唯一的 Tailwind 入口）
@import "tailwindcss";  /* ✅ v4 的唯一导入 */

@theme {
  /* 在这里覆盖 / 新增设计令牌 */
  --color-brand: #da373d;
  --font-display: "Inter", "Noto Sans SC", system-ui, sans-serif;
  --radius-card: 12px;
}

/* 普通 CSS 可以放在 @theme 之后，也可以用 var() 消费 token */
html, body, #root { height: 100%; }

✅ v4 vs v3 构建链变化
· 不需要 postcss.config.js
· 不需要 tailwind.config.js
· tailwind.config.js 里的全部配置 → 搬到 src/index.css 的 @theme { } 里
· content: ["./src/**/*.{tsx,html}"] → 默认已自动扫描，或用 @source 指定
